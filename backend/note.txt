# 4. Database

## `db/schema.sql`

```sql
CREATE DATABASE IF NOT EXISTS `gpt-clone`;

USE `gpt-clone`;

CREATE TABLE IF NOT EXISTS conversations (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    role ENUM('user', 'assistant') NOT NULL,
    content TEXT NOT NULL,
    token_count INT UNSIGNED NOT NULL DEFAULT 0,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (id)
);
```

```sql
USE `gpt-clone`;

DESCRIBE conversations;
```

```js
const errorHandler = (err, req, res, next) => {
  console.error("ERROR:", err.message);

  if (res.headersSent) {
    return next(err);
  }

  const statusCode = err.statusCode || 500;

  res.status(statusCode).json({
    status: false,
    message:
      statusCode === 500
        ? "Something went wrong try again later"
```

# 13. Chat service

This is the **main business logic**.

## `src/api/chat/service/chat.service.js`

```js
const db = require("../../../../db/db.config");

const {
  geminiClient,
  geminiModel
} = require("../../../utils/gemini");

const SYSTEM_PROMPT =
  require("../../../utils/system-prompt");

// ======================================================
// Get previous conversation history
// ======================================================

const getConversationRows = async (limit = 5) => {
  const safeLimit = Math.max(
    1,
    Math.min(Number(limit) || 5, 100)
  );

  const [rows] = await db.execute(
    `SELECT
        id,
        role,
        content,
        token_count,
        created_at
     FROM conversations
     ORDER BY id DESC
     LIMIT ${safeLimit}`
  );

  return rows.reverse();
};

// ======================================================
// Get recent conversations
// ======================================================

const getRecentConversationRows = async (limit = 100) => {
  const safeLimit = Math.max(
    1,
    Math.min(Number(limit) || 100, 100)
  );

  const [rows] = await db.execute(
    `SELECT
        id,
        role,
        content,
        token_count,
        created_at
     FROM conversations
     ORDER BY id DESC
     LIMIT ${safeLimit}`
  );

  return rows.reverse();
};

// ======================================================
// Generate Gemini answer
// ======================================================

const generateAssistantAnswer = async ({
  history,
  question
}) => {
  const contents = [];

  // Previous messages
  for (const message of history) {
    contents.push({
      role:
        message.role === "assistant"
          ? "model"
          : "user",

      parts: [
        {
          text: message.content
        }
      ]
    });
  }

  // New question
  contents.push({
    role: "user",

    parts: [
      {
        text: question
      }
    ]
  });

  const response =
    await geminiClient.models.generateContent({
      model: geminiModel,

      contents,

      config: {
        systemInstruction: SYSTEM_PROMPT
      }
    });

  const answer =
    typeof response.text === "function"
      ? response.text()
      : response.text;

  const cleanedAnswer =
    typeof answer === "string"
      ? answer.trim()
      : "";

  if (!cleanedAnswer) {
    const error = new Error(
      "The model returned an empty answer"
    );

    error.statusCode = 500;

    throw error;
  }

  const tokenCount =
    response.usageMetadata?.totalTokenCount || 0;

  return {
    content: cleanedAnswer,
    tokenCount
  };
};

// ======================================================
// Create conversation
// ======================================================

const createConversationService = async (
  question
) => {
  // 1. Validate
  if (
    typeof question !== "string" ||
    !question.trim()
  ) {
    const error = new Error(
      "question is required"
    );

    error.statusCode = 400;

    throw error;
  }

  // 2. Normalize
  const normalizedQuestion =
    question.trim();

  // 3. Maximum length
  if (normalizedQuestion.length > 65535) {
    const error = new Error(
      "question is too long"
    );

    error.statusCode = 400;

    throw error;
  }

  // 4. Get previous 5 messages
  const history =
    await getConversationRows(5);

  // 5. Save user message
  const [userResult] =
    await db.execute(
      `INSERT INTO conversations
        (role, content, token_count)
       VALUES (?, ?, ?)`,
      [
        "user",
        normalizedQuestion,
        0
      ]
    );

  const userId =
    userResult.insertId;

  // 6. Generate AI answer
  const assistant =
    await generateAssistantAnswer({
      history,
      question: normalizedQuestion
    });

  // 7. Save assistant answer
  const [assistantResult] =
    await db.execute(
      `INSERT INTO conversations
        (role, content, token_count)
       VALUES (?, ?, ?)`,
      [
        "assistant",
        assistant.content,
        assistant.tokenCount
      ]
    );

  const assistantId =
    assistantResult.insertId;

  // 8. Fetch newly created records
  const [rows] =
    await db.execute(
      `SELECT
          id,
          role,
          content,
          token_count,
          created_at
       FROM conversations
       WHERE id IN (?, ?)
       ORDER BY id ASC`,
      [
        userId,
        assistantId
      ]
    );

  const userRow =
    rows.find(
      (row) => row.id === userId
    );

  const assistantRow =
    rows.find(
      (row) => row.id === assistantId
    );

  return {
    userConversation:
      formatConversation(userRow),

    assistantConversation:
      formatConversation(assistantRow)
  };
};

// ======================================================
// Format database row
// ======================================================

const formatConversation = (row) => {
  if (!row) {
    return null;
  }

  return {
    id: row.id,
    role: row.role,
    content: row.content,
    tokenCount: row.token_count,
    createdAt: row.created_at
  };
};

module.exports = {
  getConversationRows,
  getRecentConversationRows,
  generateAssistantAnswer,
  createConversationService,
  formatConversation
};
```

# 14. Chat controller

## `src/api/chat/controller/chat.controller.js`

```js
const {
  createConversationService,
  getRecentConversationRows,
  formatConversation
} = require("../service/chat.service");

// ======================================================
// POST controller
// ======================================================

const createConversationController = async (
  req,
  res,
  next
) => {
  try {
    const { question } = req.body;

    const result =
      await createConversationService(
        question
      );

    res.status(201).json({
      success: true,
      message:
        "Conversation posted successfully.",

      data: result
    });
  } catch (error) {
    next(error);
  }
};

// ======================================================
// GET controller
// ======================================================

const getConversationController = async (
  req,
  res,
  next
) => {
  try {
    const rows =
      await getRecentConversationRows(100);

    const conversations =
      rows.map(formatConversation);

    res.status(200).json({
      success: true,

      message:
        "Conversations fetched successfully.",

      data: {
        conversations
      }
    });
  } catch (error) {
    next(error);
  }
};

module.exports = {
  createConversationController,
  getConversationController
};
```

# 28. POST flow

When the user sends a question:

```text
POST
 ↓
validate question
 ↓
get latest 5 previous messages
 ↓
save user question
 ↓
create Gemini context
 ↓
send to Gemini
 ↓
receive answer
 ↓
save assistant answer
 ↓
return both records
```

The **5-message history** is for AI context.

The **100-message history** is for displaying chat history.
